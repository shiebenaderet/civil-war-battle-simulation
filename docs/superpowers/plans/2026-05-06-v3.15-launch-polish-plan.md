- [ ] **Step 1: Create `js/settings.js`**

```javascript
(function() {
    'use strict';
    const STORAGE_KEY = 'cwSettings_v1';
    const DEFAULTS = {
        readingLevel: 'intermediate',
        fontFamily: 'serif',
        fontSize: 'medium',
        ttsVoice: null,
        ttsRate: 1.0
    };

    function load() {
        try {
            const raw = localStorage.getItem(STORAGE_KEY);
            if (!raw) return Object.assign({}, DEFAULTS);
            return Object.assign({}, DEFAULTS, JSON.parse(raw));
        } catch (e) {
            console.warn('Settings load failed; using defaults', e);
            return Object.assign({}, DEFAULTS);
        }
    }

    function save(settings) {
        try { localStorage.setItem(STORAGE_KEY, JSON.stringify(settings)); }
        catch (e) { console.warn('Settings save failed', e); }
    }

    function apply(settings) {
        const html = document.documentElement;
        html.setAttribute('data-font-family', settings.fontFamily);
        html.setAttribute('data-font-size', settings.fontSize);
    }

    const Settings = {
        load: load,
        save: save,
        apply: apply,
        get: function(key) { return load()[key]; },
        set: function(key, value) {
            const s = load();
            s[key] = value;
            save(s);
            apply(s);
            return s;
        },
        getAll: load
    };

    apply(load());
    window.Settings = Settings;
})();
```

function initNavbarReadingPills() {
    const pills = document.querySelectorAll('.reading-pills .reading-pill');
    pills.forEach(function(pill) {
        pill.addEventListener('click', function() {
            const level = pill.getAttribute('data-level');
            setReadingLevelEverywhere(level);
        });
    });
}

function setReadingLevelEverywhere(level) {
    if (level !== 'beginner' && level !== 'intermediate' && level !== 'advanced') return;

    if (typeof gameState !== 'undefined') gameState.difficulty = level;
    if (typeof Settings !== 'undefined') Settings.set('readingLevel', level);

```javascript
(function() {
    'use strict';

    let allVoices = [];
    let usableVoices = [];
    let currentUtterance = null;
    let initialized = false;

    function isUsable(voice) {
        if (!voice || !voice.lang) return false;
        if (!voice.lang.toLowerCase().startsWith('en')) return false;
        const name = (voice.name || '').toLowerCase();
        if (name.indexOf('espeak') !== -1) return false;
        return true;
    }

    function refreshVoices() {
        if (!('speechSynthesis' in window)) return;
        allVoices = window.speechSynthesis.getVoices();
        usableVoices = allVoices.filter(isUsable);
    }

    function init() {
        if (initialized) return;
        if (!('speechSynthesis' in window)) { initialized = true; return; }
        refreshVoices();
        if (window.speechSynthesis.onvoiceschanged !== undefined) {
            window.speechSynthesis.onvoiceschanged = function() {
                refreshVoices();
                document.dispatchEvent(new CustomEvent('tts:voices-updated'));
            };
        }
        initialized = true;
    }

    function getVoices() { return usableVoices.slice(); }
    function isAvailable() { return usableVoices.length > 0; }

    function qualityWarning() {
        if (!isAvailable()) return null;
        const allRemote = usableVoices.every(function(v) { return v.localService === false; });
        if (allRemote) return 'Quality varies by device.';
        const allGeneric = usableVoices.every(function(v) { return /^(en|english)$/i.test(v.name || ''); });
        if (allGeneric) return 'Quality varies by device.';
        return null;
    }

    function getVoiceByName(name) {
        for (let i = 0; i < usableVoices.length; i++) {
            if (usableVoices[i].name === name) return usableVoices[i];
        }
        return null;
    }

    function speak(text, opts) {
        if (!isAvailable()) return;
        opts = opts || {};
        stop();
        const u = new SpeechSynthesisUtterance(text);
        const voiceName = opts.voice || (window.Settings && Settings.get('ttsVoice'));
        const voice = voiceName ? getVoiceByName(voiceName) : null;
        if (voice) u.voice = voice;
        u.rate = opts.rate || (window.Settings && Settings.get('ttsRate')) || 1.0;
        u.pitch = 1.0;
        u.onend = function() {
            currentUtterance = null;
            if (typeof opts.onEnd === 'function') opts.onEnd();
        };
        u.onerror = function() {
            currentUtterance = null;
            if (typeof opts.onError === 'function') opts.onError();
        };
        currentUtterance = u;
        window.speechSynthesis.speak(u);
    }

    function stop() {
        if ('speechSynthesis' in window) window.speechSynthesis.cancel();
        currentUtterance = null;
    }

    function isSpeaking() {
        return currentUtterance !== null && window.speechSynthesis.speaking;
    }

    init();

    window.TTS = {
        init: init, getVoices: getVoices, isAvailable: isAvailable,
        qualityWarning: qualityWarning, speak: speak, stop: stop, isSpeaking: isSpeaking
    };
})();

function attachTtsButton(targetEl, label) {
    if (!targetEl) return null;

        e.preventDefault();
        e.stopPropagation();
        if (btn.getAttribute('data-playing') === 'true') {
            TTS.stop();
            return;
        }
        document.querySelectorAll('.tts-play-btn[data-playing="true"]').forEach(function(other) {
            other.setAttribute('data-playing', 'false');
            other.textContent = '▶ Read';
        });
        const text = Array.from(targetEl.childNodes)
            .filter(function(n) { return n !== btn; })
            .map(function(n) { return n.textContent || ''; })
            .join(' ').trim();
        if (!text) return;

                btn.setAttribute('data-playing', 'false');

        });
    });

        if (s === null || s === undefined) return '';
        return String(s).replace(/[&<>"']/g, function(c) {
            return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c];
        });
    }

    function getOptionText(battle, optIdx, level, side) {
        try {
            const sideKey = side === 'union' ? 'union' : 'confederacy';
            const opts = battle.historical.whatWouldYouDo[sideKey].options[level];
            return opts[optIdx] || '';
        } catch (e) { return ''; }
    }

    function buildHtml() {
        const studentName = (typeof gameState !== 'undefined' && gameState.userName) || 'Student';
        const side = (typeof gameState !== 'undefined' && gameState.userSide) || 'Unknown';
        const level = (typeof gameState !== 'undefined' && gameState.difficulty) || 'intermediate';
        const completedDate = new Date().toLocaleDateString();

        let html = '<!DOCTYPE html><html><head><meta charset="UTF-8"><title>Civil War Simulation Summary - ' + escape(studentName) + '</title>';
        html += '<style>';
        html += 'body{font-family:Georgia,serif;line-height:1.5;max-width:7in;margin:0.5in auto;color:#000;}';
        html += 'h1{font-size:18pt;border-bottom:2px solid #000;padding-bottom:6px;margin-bottom:12px;}';
        html += 'h2{font-size:14pt;margin-top:24px;border-bottom:1px solid #000;padding-bottom:4px;page-break-before:always;}';
        html += 'h2:first-of-type{page-break-before:auto;}';
        html += 'h3{font-size:12pt;margin-top:14px;}';
        html += 'p,li{font-size:10.5pt;}';
        html += 'table{border-collapse:collapse;width:100%;margin:8px 0;}';
        html += 'th,td{border:1px solid #888;padding:4px 6px;font-size:9.5pt;text-align:left;vertical-align:top;}';
        html += 'th{background:#eee;}';
        html += '.meta{font-size:9pt;color:#444;margin-bottom:18px;}';
        html += '.reflection{border-left:3px solid #6b1a17;padding:6px 12px;margin:8px 0;background:#faf6ec;font-size:10.5pt;white-space:pre-wrap;}';
        html += '@media print{body{margin:0.5in;}}';
        html += '</style>';
        html += '<script>window.addEventListener("load",function(){setTimeout(function(){window.print();},250);});<\/script>';
        html += '</head><body>';

        html += '<h1>Civil War Simulation</h1>';
        html += '<div class="meta">';
        html += '<strong>Student:</strong> ' + escape(studentName) + '<br>';
        html += '<strong>Side:</strong> ' + escape(side.charAt(0).toUpperCase() + side.slice(1)) + '<br>';
        html += '<strong>Reading level:</strong> ' + escape(level.charAt(0).toUpperCase() + level.slice(1)) + '<br>';
        html += '<strong>Completed:</strong> ' + escape(completedDate);
        html += '</div>';

        if (typeof acts !== 'undefined' && Array.isArray(acts)) {
            acts.forEach(function(act, actIdx) {
                html += '<h2>Act ' + escape(act.number) + ' — ' + escape(act.name) + ' (' + escape(act.years) + ')</h2>';

                html += '<h3>Battles</h3>';
                html += '<table><thead><tr><th>Battle</th><th>Date</th><th>Your Decision</th><th>Historical Decision</th><th>Match</th></tr></thead><tbody>';
                act.battleIndices.forEach(function(bIdx) {
                    const battle = battles[bIdx];
                    const progress = (gameState.battles && gameState.battles[bIdx]) || {};
                    const userOpt = (progress.wwydOptionIndex !== undefined && progress.wwydOptionIndex !== null) ? progress.wwydOptionIndex : null;
                    const userText = (userOpt !== null) ? getOptionText(battle, userOpt, level, gameState.userSide) : 'Not played';
                    const histText = getOptionText(battle, 0, level, gameState.userSide) || '—';
                    const match = (userOpt === 0) ? 'Yes' : (userOpt === null ? '—' : 'No');
                    html += '<tr><td>' + escape(battle.name) + '</td><td>' + escape(battle.date) + '</td><td>' + escape(userText) + '</td><td>' + escape(histText) + '</td><td>' + escape(match) + '</td></tr>';
                });
                html += '</tbody></table>';

                if (act.recallQuestions && act.recallQuestions[level]) {
                    html += '<h3>Recall Questions</h3>';
                    html += '<table><thead><tr><th>Question</th><th>Your Answer</th><th>Correct Answer</th></tr></thead><tbody>';
                    const recall = (gameState.completedRecalls && gameState.completedRecalls[actIdx]) || {};
                    act.recallQuestions[level].forEach(function(q, qIdx) {
                        const userIdx = recall[qIdx];
                        const userAns = (userIdx !== undefined && q.options[userIdx]) ? q.options[userIdx] : 'Not answered';
                        const correctAns = q.options[q.correctIndex] || '';
                        html += '<tr><td>' + escape(q.question) + '</td><td>' + escape(userAns) + '</td><td>' + escape(correctAns) + '</td></tr>';
                    });
                    html += '</tbody></table>';
                }

                const reflection = (gameState.reflections && gameState.reflections[actIdx]) || '';
                if (reflection) {
                    html += '<h3>Your Reflection</h3>';
                    html += '<div class="reflection">' + escape(reflection) + '</div>';
                }
            });
        }

        html += '</body></html>';
        return html;
    }

    function generate() {
        const html = buildHtml();
        const blob = new Blob([html], { type: 'text/html' });
        const url = URL.createObjectURL(blob);
        const w = window.open(url, '_blank');
        if (!w) {
            alert('Could not open print window. Please allow popups, or use your browser File > Print.');
            URL.revokeObjectURL(url);
            return;
        }
        // Revoke the URL after a generous delay so the new window finishes loading.
        setTimeout(function() { URL.revokeObjectURL(url); }, 60000);
    }

    window.PrintSummary = { generate: generate };
})();
```
